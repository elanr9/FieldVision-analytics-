const { SectionHeading, Card, StatCard, SegmentedControl } = window.InkboundDesignSystem_74e23a;
function Spark({ days, color = 'var(--ink-700)', h = 28, w = 96 }) {
  const max = Math.max(1, ...days); const pts = days.map((v, i) => (i / (days.length - 1) * w) + ',' + (h - v / max * (h - 2) - 1)).join(' ');
  return <svg width={w} height={h} viewBox={`0 0 ${w} ${h}`} style={{ display: 'block' }}><polyline points={pts} fill="none" stroke={color} strokeWidth="1.5" strokeLinejoin="round" style={{ strokeDasharray: 400, strokeDashoffset: 400, animation: 'ink-draw 800ms var(--ease-out) forwards' }} /></svg>;
}
function Delta({ v }) { const up = v >= 0; return <span style={{ font: '600 11px/1.2 var(--font-sans)', color: up ? 'var(--green-800)' : 'var(--red-800)', background: up ? 'var(--green-100)' : 'var(--red-100)', borderRadius: 9999, padding: '2px 7px', fontVariantNumeric: 'tabular-nums' }}>{up ? '+' : ''}{v}%</span>; }
function UsageScreen({ D, focus }) {
  const { pop } = React.useContext(NavCtx);
  const [metric, setMetric] = React.useState('users');
  const [sel, setSel] = React.useState(focus || D.usage[0].id);
  const [view, setView] = React.useState(focus ? 'trend' : 'ranked');
  const sorted = [...D.usage].sort((a, b) => metric === 'users' ? b.users30 - a.users30 : metric === 'events' ? b.events30 - a.events30 : b.perUser - a.perUser);
  const key = metric === 'users' ? 'users30' : metric === 'events' ? 'events30' : 'perUser';
  const max = sorted[0][key];
  const totalEvents = D.usage.reduce((s, f) => s + f.events30, 0);
  const active = Math.max(...D.usage.map(f => f.users30));
  const least = [...D.usage].sort((a, b) => a.users30 - b.users30)[0];
  const selF = sel ? D.usage.find(f => f.id === sel) : null;
  const dayLabels = Array.from({ length: 30 }, (_, i) => D.fmt(D.day(29 - i)));
  return (
    <main style={{ maxWidth: 768, margin: '0 auto', padding: '0 16px 64px', fontFamily: 'var(--font-sans)' }}>
      <SubHeader title="Feature usage" onBack={pop} />
      <div style={{ paddingTop: 16, display: 'grid', gap: 12, gridTemplateColumns: 'minmax(0,1fr)' }}>
        <SegmentedControl value={view} onChange={setView} options={[{ key: 'ranked', label: 'Ranked' }, { key: 'trend', label: 'Trend' }, { key: 'share', label: 'Share' }]} />
        <Fade id={view}>
        {view === 'ranked' && <section>
          <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', marginBottom: 8 }}><SectionHeading style={{ marginBottom: 0 }}>Last 30 days</SectionHeading><MiniToggle value={metric} onChange={setMetric} options={[['users', 'Users'], ['events', 'Events'], ['perUser', 'Per user']]} /></div>
          <Card padding="none">
            {sorted.map((f, i) => { const on = sel === f.id; return (
              <button key={f.id} type="button" onClick={() => setSel(on ? null : f.id)} style={{ display: 'grid', gridTemplateColumns: '18px 1fr 80px auto', alignItems: 'center', gap: 10, width: '100%', padding: '11px 14px', background: on ? 'var(--ink-50)' : 'transparent', border: 0, borderTop: i ? '1px solid var(--border-subtle)' : 0, cursor: 'pointer', textAlign: 'left', fontFamily: 'var(--font-sans)', color: 'var(--text-primary)', transition: 'background-color var(--duration-fast)' }}>
                <FIcon id={f.id} size={18} color={on ? 'var(--ink-900)' : 'var(--ink-600)'} />
                <span style={{ minWidth: 0 }}>
                  <span style={{ display: 'flex', alignItems: 'center', gap: 8 }}><span style={{ font: '600 14px/1.3 var(--font-sans)' }}>{f.label}</span><Delta v={f.delta} /></span>
                  <span style={{ display: 'block', marginTop: 6, height: 6, borderRadius: 3, background: 'var(--surface-track)', overflow: 'hidden' }}><span style={{ display: 'block', height: '100%', width: (f[key] / max * 100) + '%', background: 'var(--ink-700)', borderRadius: 3, transformOrigin: '0 0', animation: 'ink-grow-x 600ms var(--ease-out) both' }} /></span>
                </span>
                <Spark days={f.days} w={80} />
                <span style={{ textAlign: 'right', minWidth: 56 }}><span style={{ display: 'block', font: '700 16px/1.2 var(--font-sans)', fontVariantNumeric: 'tabular-nums' }}><CountUp value={f[key]} decimals={metric === 'perUser' ? 1 : 0} /></span><span style={{ font: '400 10px/1.4 var(--font-sans)', color: 'var(--text-tertiary)', textTransform: 'uppercase', letterSpacing: '.03em' }}>{metric === 'users' ? 'users' : metric === 'events' ? 'events' : 'per user'}</span></span>
              </button>); })}
          </Card>
        </section>}
        {view === 'trend' && selF && <section key={selF.id}>
          <div style={{ display: 'flex', gap: 6, overflowX: 'auto', paddingBottom: 8, margin: '0 -16px', padding: '0 16px 8px' }}>{[...D.usage].sort((a, b) => b.events30 - a.events30).map(f => <button key={f.id} type="button" onClick={() => setSel(f.id)} style={{ flexShrink: 0, display: 'flex', alignItems: 'center', gap: 6, border: 0, borderRadius: 9999, padding: '6px 12px', cursor: 'pointer', font: '600 12px/1.4 var(--font-sans)', background: sel === f.id ? 'var(--ink-700)' : 'var(--gray-100)', color: sel === f.id ? '#fff' : 'var(--gray-600)' }}><FIcon id={f.id} size={14} color="currentColor" />{f.label}</button>)}</div>
          <Hero label={selF.label + ' · 30 days'} value={selF.events30} caption={selF.desc + ' · ' + selF.users30 + ' users · ' + selF.perUser + ' per user'} right={<Delta v={selF.delta} />} />
          <Card padding="wide" style={{ marginTop: 12 }}>
            <InkLineChart data={selF.days.map((v, i) => ({ label: dayLabels[i], v }))} series={[{ key: 'v', color: 'var(--ink-700)' }]} height={170} />
          </Card>
          <div style={{ display: 'grid', gridTemplateColumns: 'repeat(3,minmax(0,1fr))', gap: 8, marginTop: 12 }}><Mini label="Best day" value={Math.max(...selF.days)} sub="events" /><Mini label="Avg / day" value={Math.round(selF.events30 / 30)} sub="events" /><Mini label="Last 15d" value={(selF.delta >= 0 ? '+' : '') + selF.delta + '%'} sub="vs prior 15d" /></div>
        </section>}
        {view === 'share' && <section>
          <div style={{ display: 'grid', gridTemplateColumns: 'repeat(3,minmax(0,1fr))', gap: 8, marginBottom: 12 }}>
            <Mini label="Events" value={totalEvents} sub={Math.round(totalEvents / 30) + ' per day'} />
            <Mini label="Active users" value={active} sub="any feature" />
            <Mini label="Least used" value={least.users30} sub={least.label} />
          </div>
          <Card padding="wide">
            <div style={{ display: 'flex', height: 12, borderRadius: 6, overflow: 'hidden', gap: 2 }}>{[...D.usage].sort((a, b) => b.events30 - a.events30).map((f, i) => <span key={f.id} style={{ width: (f.events30 / totalEvents * 100) + '%', background: ['var(--ink-900)', 'var(--ink-700)', 'var(--ink-500)', 'var(--ink-400)', 'var(--ink-300)', 'var(--ink-200)', 'var(--gray-300)', 'var(--gray-200)'][i] }} />)}</div>
            <div style={{ display: 'flex', flexWrap: 'wrap', gap: '6px 16px', marginTop: 12 }}>{[...D.usage].sort((a, b) => b.events30 - a.events30).map((f, i) => <span key={f.id} style={{ display: 'flex', alignItems: 'center', gap: 6, font: '400 12px/1.5 var(--font-sans)', color: 'var(--gray-600)' }}><span style={{ width: 8, height: 8, borderRadius: 2, background: ['var(--ink-900)', 'var(--ink-700)', 'var(--ink-500)', 'var(--ink-400)', 'var(--ink-300)', 'var(--ink-200)', 'var(--gray-300)', 'var(--gray-200)'][i] }} />{f.label} <span style={{ color: 'var(--text-tertiary)', fontVariantNumeric: 'tabular-nums' }}>{Math.round(f.events30 / totalEvents * 100)}%</span></span>)}</div>
          </Card>
        </section>}
        </Fade>
      </div>
    </main>
  );
}
Object.assign(window, { UsageScreen, Spark, Delta });
