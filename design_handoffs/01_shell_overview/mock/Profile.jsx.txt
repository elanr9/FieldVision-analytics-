const { StatusBadge, Tag, ContactActions, SectionHeading, StatBlock, Card, Skeleton, Button, SegmentedControl, MetaRow } = window.InkboundDesignSystem_74e23a;
/** One coach conversation, as it looks in the athlete's Inkbound inbox. */
function ConversationScreen({ D, user, reply }) {
  const { pop } = React.useContext(NavCtx);
  const msgs = D.thread(user, reply);
  return (
    <main style={{ maxWidth: 768, margin: '0 auto', padding: '0 16px 32px', fontFamily: 'var(--font-sans)' }}>
      <SubHeader title={reply.school} onBack={pop} />
      <div style={{ paddingTop: 16 }}>
        <p style={{ margin: 0, font: '600 10px/1.4 var(--font-sans)', textTransform: 'uppercase', letterSpacing: '.04em', color: 'var(--text-secondary)' }}>Subject</p>
        <p style={{ margin: '2px 0 0', font: '600 14px/1.4 var(--font-sans)' }}>{msgs[0].subject}</p>
        <p style={{ margin: '2px 0 0', font: '400 12px/1.5 var(--font-sans)', color: 'var(--text-secondary)' }}>{user.name} ↔ {reply.coach} · {reply.school}</p>
        <div style={{ display: 'grid', gap: 10, marginTop: 16 }}>
          {msgs.map((m, i) => { const me = m.from === 'athlete'; return (
            <div key={i} style={{ display: 'flex', flexDirection: 'column', alignItems: me ? 'flex-end' : 'flex-start', animation: `ink-fade var(--duration-base) var(--ease-out) ${i * 60}ms both` }}>
              <div style={{ maxWidth: '86%', background: me ? 'var(--ink-700)' : 'var(--surface-card)', color: me ? '#fff' : 'var(--text-primary)', border: me ? 0 : '1px solid var(--border-default)', borderRadius: me ? '16px 16px 4px 16px' : '16px 16px 16px 4px', padding: '10px 14px', font: '400 14px/1.5 var(--font-sans)', whiteSpace: 'pre-line' }}>{m.text}</div>
              <span style={{ marginTop: 3, font: '400 10px/1.4 var(--font-sans)', color: 'var(--text-tertiary)' }}>{me ? user.name.split(' ')[0] : reply.coach} · {D.fmt(D.day(m.daysAgo))} · {D.ago(m.daysAgo)}</span>
            </div>); })}
        </div>
      </div>
    </main>
  );
}
function Fact({ label, value }) { return <div><p style={{ margin: 0, font: '600 10px/1.4 var(--font-sans)', textTransform: 'uppercase', letterSpacing: '.03em', color: 'var(--text-secondary)' }}>{label}</p><p style={{ margin: '2px 0 0', font: '600 14px/1.4 var(--font-sans)' }}>{value}</p></div>; }
function ProfileScreen({ D, user, sent }) {
  const { pop, text, open } = React.useContext(NavCtx);
  const athlete = user.isParent && D.byId[user.athleteId];
  const [loading, setLoading] = React.useState(true);
  const [sec, setSec] = React.useState('background');
  const [chap, setChap] = React.useState('basic');
  const { push } = React.useContext(NavCtx);
  React.useEffect(() => { const t = setTimeout(() => setLoading(false), 700); return () => clearTimeout(t); }, [user.id]);
  const facts = [user.team, user.position, user.gradYear ? "'" + String(user.gradYear).slice(-2) : null].filter(Boolean).join(' · ');
  const planFact = user.status === 'trialing' ? [user.planLabel + ' · trial', 'Trial ends ' + D.fmt(D.day(-Math.ceil((user.trialEndsIn || 0) / 24))) + ' · in ' + user.trialEndsIn + 'h'] : user.status === 'paying' ? [user.planLabel, 'Next charge ' + D.fmt(D.day(-user.nextCharge))] : user.status === 'churned' ? [user.planLabel + ' · cancelled', 'Cancelled ' + D.ago(user.cancelledAgo)] : user.status === 'trial_ended' ? [user.planLabel + ' · trial', 'Trial ended ' + D.ago(user.trialEndedAgo)] : user.onboarding === 'in_progress' ? ['No plan', 'In onboarding · ' + user.stepId] : ['No plan', user.stoppedAtPaywall ? 'Stopped at paywall' : 'Never started onboarding'];
  const checkinEligible = (user.status === 'paying' || user.status === 'trialing') && user.phone;
  const wasSent = sent[user.id];
  const lastCheckin = wasSent ? 'today' : user.lastCheckin == null ? 'never' : D.ago(user.lastCheckin);
  if (user.isParent) return (
    <main style={{ maxWidth: 768, margin: '0 auto', padding: '0 16px 64px', fontFamily: 'var(--font-sans)' }}>
      <SubHeader title={user.name} onBack={pop} />
      <div style={{ paddingTop: 20 }}>
        <h2 style={{ margin: 0, font: '700 26px/1.2 var(--font-sans)', letterSpacing: '-0.015em' }}>{user.name}</h2>
        <p style={{ margin: '4px 0 0', font: '400 14px/1.5 var(--font-sans)', color: 'var(--text-secondary)' }}>{[user.email || 'No email', user.phone].filter(Boolean).join(' · ')}</p>
        <div style={{ marginTop: 10, display: 'flex', gap: 6 }}><Tag kind="soft">Parent</Tag><span style={{ borderRadius: 9999, padding: '2px 8px', font: '600 11px/1.5 var(--font-sans)', background: user.parentActive ? 'var(--green-100)' : 'var(--gray-100)', color: user.parentActive ? 'var(--green-800)' : 'var(--text-secondary)' }}>{user.parentActive ? 'Opened the app' : 'Never opened the app'}</span></div>
        <Card padding="wide" style={{ marginTop: 16, display: 'grid', gridTemplateColumns: '1fr 1fr', gap: 16 }}>
          <Fact label="Joined" value={user.joined} />
          <Fact label="Invited by" value={athlete ? athlete.name.split(' ')[0] + ' · onboarding' : '—'} />
        </Card>
        <div style={{ marginTop: 12 }}><ContactActions phone={user.phone} email={user.email} size="lg" /></div>
        <SectionHeading style={{ marginTop: 28 }}>Their athlete</SectionHeading>
        {athlete ? <Card interactive padding="wide" onClick={() => open(athlete)} style={{ display: 'grid', gridTemplateColumns: '1fr auto', alignItems: 'center', gap: 12 }}>
          <div style={{ minWidth: 0 }}>
            <p style={{ margin: 0, font: '700 16px/1.3 var(--font-sans)' }}>{athlete.name}</p>
            <p style={{ margin: '2px 0 0', font: '400 12px/1.5 var(--font-sans)', color: 'var(--text-secondary)' }}>{[athlete.team, athlete.position, athlete.gradYear ? "'" + String(athlete.gradYear).slice(-2) : null].filter(Boolean).join(' · ') || 'No team on file'}</p>
            <div style={{ marginTop: 8, display: 'flex', gap: 6, alignItems: 'center' }}><StatusBadge status={athlete.status} />{athlete.planLabel && <span style={{ font: '400 12px var(--font-sans)', color: 'var(--text-secondary)' }}>{athlete.planLabel}</span>}</div>
          </div>
          <span style={{ color: 'var(--text-tertiary)', fontSize: 22, lineHeight: 1 }}>›</span>
        </Card> : <p style={{ margin: 0, font: '400 14px/1.5 var(--font-sans)', color: 'var(--text-tertiary)' }}>No athlete linked.</p>}
        {athlete && <div style={{ display: 'grid', gridTemplateColumns: 'repeat(3,minmax(0,1fr))', gap: 8, marginTop: 8 }}><Mini label="Emails" value={athlete.stats.emails} sub="to coaches" /><Mini label="Replies" value={athlete.stats.replies} sub="from coaches" /><Mini label="Videos" value={athlete.videos.length} sub="made" /></div>}
      </div>
    </main>
  );
  return (
    <main style={{ maxWidth: 768, margin: '0 auto', padding: '0 16px 64px', fontFamily: 'var(--font-sans)' }}>
      <SubHeader title={user.name} onBack={pop} />
      <div style={{ paddingTop: 20 }}>
        <h2 style={{ margin: 0, font: '700 26px/1.2 var(--font-sans)', letterSpacing: '-0.015em' }}>{user.name}</h2>
        <p style={{ margin: '4px 0 0', font: '400 14px/1.5 var(--font-sans)', color: 'var(--text-secondary)' }}>{[user.email || 'No email', user.phone].filter(Boolean).join(' · ')}</p>
        <div style={{ marginTop: 10, display: 'flex', flexWrap: 'wrap', alignItems: 'center', gap: 6 }}><StatusBadge status={user.status} />{user.isParent && <button type="button" onClick={() => athlete && open(athlete)} style={{ border: 0, background: 'var(--ink-50)', color: 'var(--ink-700)', borderRadius: 9999, padding: '2px 8px', font: '600 11px/1.5 var(--font-sans)', cursor: athlete ? 'pointer' : 'default' }}>Parent of {athlete ? athlete.name : '—'} ›</button>}{user.excludedFromMetrics && <Tag kind="soft">Internal</Tag>}{user.junk && <Tag kind="soft">Likely fake · {user.junkReason}</Tag>}</div>
        <Card padding="wide" style={{ marginTop: 12, display: 'grid', gridTemplateColumns: '1fr 1fr', gap: 12, paddingTop: 12, paddingBottom: 12 }}>
          <Fact label="Plan" value={planFact[0]} />
          <Fact label={user.status === 'paying' ? 'Billing' : 'Status'} value={planFact[1]} />
          {facts && <Fact label="Team · position · grad" value={facts} />}
          {checkinEligible && <Fact label="Last check-in" value={lastCheckin} />}
        </Card>
        <div style={{ marginTop: 12 }}><ContactActions phone={user.phone} email={user.email} size="lg" onText={checkinEligible ? () => text(user) : undefined} /></div>
        <div style={{ display: 'grid', gridTemplateColumns: 'repeat(4,1fr)', gap: 8, marginTop: 10 }}><StatBlock label="Emails" value={<CountUp value={user.stats.emails} />} /><StatBlock label="Replies" value={<CountUp value={user.stats.replies} />} /><StatBlock label="HL views" value={<CountUp value={user.stats.views} />} /><StatBlock label="Calls" value={<CountUp value={user.stats.calls} />} /></div>
        <SegmentedControl style={{ marginTop: 12, marginBottom: 10 }} value={sec} onChange={setSec} options={[{ key: 'background', label: 'Background' }, { key: 'videos', label: 'Videos · ' + user.videos.length }, { key: 'replies', label: 'Replies · ' + user.replies.length }]} />
        <Fade id={sec + chap}>
        {sec === 'background' && (() => { const B = D.background(user), ch = B.find(b => b.key === chap) || B[0]; return <div style={{ display: 'grid', gap: 10 }}>
          <MiniToggle value={chap} onChange={setChap} options={[['basic', 'Background'], ['checkin', 'Where'], ['academic', 'Academics'], ['athletic', 'Game'], ['goals', 'Goals']]} />
          <Card padding="wide" style={{ display: 'grid', gridTemplateColumns: '1fr 1fr', gap: '10px 16px', paddingTop: 12, paddingBottom: 12 }}>{ch.rows.map(([k, v]) => <div key={k} style={{ minWidth: 0 }}><p style={{ margin: 0, font: '500 10px/1.3 var(--font-sans)', textTransform: 'uppercase', letterSpacing: '.03em', color: 'var(--text-tertiary)', whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>{k}</p><p style={{ margin: '1px 0 0', font: '500 13px/1.35 var(--font-sans)', color: 'var(--text-primary)', overflow: 'hidden', textOverflow: 'ellipsis', whiteSpace: 'nowrap' }}>{v}</p></div>)}</Card>
        </div>; })()}
        {sec === 'videos' && (loading ? <Skeleton height={56} radius={12} /> : user.videos.length === 0 ? <p style={{ margin: 0, font: '400 14px/1.5 var(--font-sans)', color: 'var(--text-tertiary)' }}>No highlight videos yet.</p> :
          <Card padding="none" style={{ animation: 'ink-fade var(--duration-base) var(--ease-out) both' }}>{user.videos.map((v, i) => <div key={i} style={{ padding: '12px 16px', borderTop: i ? '1px solid var(--border-subtle)' : 0, display: 'flex', justifyContent: 'space-between', gap: 12, alignItems: 'flex-start' }}><div style={{ minWidth: 0 }}><p style={{ margin: 0, font: '600 14px/1.4 var(--font-sans)', whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>{v.title}</p><p style={{ margin: 0, font: '400 12px/1.5 var(--font-sans)', color: 'var(--text-secondary)' }}>{D.fmt(D.day(v.daysAgo))} · {D.ago(v.daysAgo)} · {v.coaches} coaches · {v.views} views</p></div><span style={{ flexShrink: 0, borderRadius: 9999, padding: '2px 8px', font: '600 11px/1.5 var(--font-sans)', background: v.status === 'Published' ? 'var(--green-100)' : 'var(--gray-100)', color: v.status === 'Published' ? 'var(--green-800)' : 'var(--gray-600)' }}>{v.status}</span></div>)}</Card>)}
        {sec === 'replies' && (loading ? <Skeleton height={72} radius={12} /> : user.replies.length === 0 ? <p style={{ margin: 0, font: '400 14px/1.5 var(--font-sans)', color: 'var(--text-tertiary)' }}>No coach replies yet.</p> :
          <Card padding="none" style={{ animation: 'ink-fade var(--duration-base) var(--ease-out) both' }}>{user.replies.map((r, i) => <button key={i} type="button" onClick={() => push({ key: 'convo-' + i, node: <ConversationScreen D={D} user={user} reply={r} /> })} style={{ display: 'block', width: '100%', textAlign: 'left', background: 'none', border: 0, cursor: 'pointer', fontFamily: 'var(--font-sans)', color: 'var(--text-primary)', padding: '12px 16px', borderTop: i ? '1px solid var(--border-subtle)' : 0 }}><div style={{ display: 'flex', justifyContent: 'space-between', gap: 12 }}><p style={{ margin: 0, font: '600 14px/1.4 var(--font-sans)' }}>{r.school} <span style={{ font: '400 12px var(--font-sans)', color: 'var(--text-tertiary)' }}>· {r.coach}</span></p><span style={{ font: '400 11px/1.4 var(--font-sans)', color: 'var(--text-tertiary)', whiteSpace: 'nowrap' }}>{D.ago(r.daysAgo)} ›</span></div><p style={{ margin: '4px 0 0', font: '400 13px/1.5 var(--font-sans)', color: 'var(--gray-700)', whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>{r.snippet}</p></button>)}</Card>)}
        </Fade>
      </div>
    </main>
  );
}
Object.assign(window, { ProfileScreen, ConversationScreen });
