export interface UserRowProps {
  user: { name: string; email?: string; phone?: string | null; team?: string | null; status: 'paying' | 'comped' | 'trialing' | 'trial_ended' | 'signed_up' | 'churned'; interval?: string; joined?: string; isParent?: boolean; excludedFromMetrics?: boolean };
