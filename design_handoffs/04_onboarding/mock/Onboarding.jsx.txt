const { SectionHeading, StatCard, Card, SegmentedControl } = window.InkboundDesignSystem_74e23a;
const RATE_LABEL = { '7d': '7d', '30d': '30d', all: 'All' };
function Rate({ label, a, b, warn }) {
  const pct = a ? Math.round(b / a * 100) : 0;
  return (
    <div style={{ display: 'grid', gridTemplateColumns: '1fr auto', alignItems: 'center', gap: 12, padding: '9px 0' }}>
      <div style={{ minWidth: 0 }}>
        <p style={{ margin: 0, font: '600 13px/1.4 var(--font-sans)' }}>{label}</p>
        <p style={{ margin: 0, font: '400 11px/1.4 var(--font-sans)', color: 'var(--text-secondary)', fontVariantNumeric: 'tabular-nums' }}>{b} of {a} people · {a - b} lost</p>
        <div style={{ marginTop: 6, height: 6, borderRadius: 3, background: 'var(--surface-track)', overflow: 'hidden' }}><div style={{ height: '100%', width: pct + '%', borderRadius: 3, background: warn && pct < 50 ? 'var(--amber-500)' : 'var(--ink-700)', transformOrigin: '0 0', animation: 'ink-grow-x 600ms var(--ease-out) both' }} /></div>
      </div>
      <p style={{ margin: 0, font: '700 20px/1 var(--font-sans)', fontVariantNumeric: 'tabular-nums', color: warn && pct < 50 ? 'var(--amber-800)' : 'inherit' }}>{pct}%</p>
    </div>
  );
}
function OnboardingTab({ D, real, range, onRange, push }) {
  const [view, setView] = React.useState('summary');
  const [sel, setSel] = React.useState(null);
  const [openCh, setOpenCh] = React.useState({});
  const [zoom, setZoom] = React.useState('all');
  const SHORT = { basic: 'Background', checkin: 'Where', academic: 'Academics', athletic: 'Game', goals: 'Goals', paywall: 'Paywall' };
  const days = range === '7d' ? 7 : range === '30d' ? 30 : 9999;
  const cohort = real.filter(u => u.signupDaysAgo < days);
  const started = cohort.filter(u => u.onboarding !== 'none'), completed = cohort.filter(u => u.onboarding === 'completed'), inProg = cohort.filter(u => u.onboarding === 'in_progress');
  const f = D.funnel, by = id => f.find(s => s.id === id), last = f[f.length - 1];
  const worst = [...f].sort((a, b) => b.dropPct - a.dropPct).slice(0, 3);
  const chapters = D.chapterKeys.map(k => { const steps = f.filter(s => s.chapter === k); return { key: k, label: D.CH[k], steps, enter: steps[0].reached + steps[0].dropped, exit: steps[steps.length - 1].reached }; });
  const people = (title, users) => push({ key: 'people-' + title, node: <PeopleScreen title={title} users={users} /> });
  const goStep = id => { setSel(id); setView('funnel'); };
  const s = sel ? by(sel) : null;
  const rangeToggle = <MiniToggle value={range} onChange={onRange} options={[['7d', '7d'], ['30d', '30d'], ['all', 'All']]} />;
  return (
    <div style={{ display: 'grid', gap: 16, gridTemplateColumns: 'minmax(0,1fr)' }}>
      <SegmentedControl value={view} onChange={setView} options={[{ key: 'summary', label: 'Summary' }, { key: 'funnel', label: 'Graph' }, { key: 'steps', label: 'Steps' }, { key: 'paywall', label: 'Paywall' }]} />
      <Fade id={view}>
        {view === 'summary' && <div style={{ display: 'grid', gap: 12 }}>
          <Hero label="Started survey → paid" value={last.pct + '%'} accent caption={last.reached + ' of ' + f[0].reached + ' people who started · last 30 days'} />
          <Card padding="wide" style={{ paddingTop: 2, paddingBottom: 2 }}>
            <Rate label="1 · Finish the survey" a={f[0].reached} b={by('account_created').reached} />
            <div style={{ borderTop: '1px solid var(--border-subtle)' }}><Rate label="2 · Start the free trial" a={by('account_created').reached} b={by('try_free').reached} warn /></div>
            <div style={{ borderTop: '1px solid var(--border-subtle)' }}><Rate label="3 · Trial → paid" a={by('try_free').reached} b={last.reached} warn /></div>
          </Card>
          <div>
            <SectionHeading>Where most people leave</SectionHeading>
            <div style={{ display: 'grid', gridTemplateColumns: 'repeat(3,minmax(0,1fr))', gap: 8 }}>{worst.map(w => <button key={w.id} type="button" onClick={() => goStep(w.id)} style={{ background: 'var(--surface-card)', border: '1px solid var(--border-default)', borderRadius: 12, padding: '10px 12px', minWidth: 0, textAlign: 'left', cursor: 'pointer', fontFamily: 'var(--font-sans)', color: 'var(--text-primary)' }}><p style={{ margin: 0, font: '600 12px/1.3 var(--font-sans)', whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>{w.label}</p><p style={{ margin: '4px 0 0', font: '700 20px/1.1 var(--font-sans)', fontVariantNumeric: 'tabular-nums', color: 'var(--amber-800)' }}>{w.dropPct}%</p><p style={{ margin: '2px 0 0', font: '400 10px/1.4 var(--font-sans)', color: 'var(--text-tertiary)', fontVariantNumeric: 'tabular-nums' }}>leave here · {w.dropped} of {w.reached + w.dropped}</p></button>)}</div>
          </div>
        </div>}
        {view === 'funnel' && (() => {
          const START = f[0].reached;
          const zc = zoom === 'all' ? null : chapters.find(c => c.key === zoom);
          const chBars = chapters.map((c, i) => ({ id: c.key, label: SHORT[c.key], pct: Math.round(c.exit / START * 1000) / 10, prevPct: i ? Math.round(chapters[i-1].exit / START * 1000) / 10 : 100, chapter: c.key === 'paywall' ? 'paywall' : 'x' }));
          const zs = zc ? zc.steps.map((st, i) => i === 0 && f.indexOf(st) > 0 ? { ...st, prevPct: f[f.indexOf(st) - 1].pct } : st) : chBars;
          const top = zc ? [...zc.steps].sort((a, b) => b.dropped - a.dropped).slice(0, 3) : null;
          const Row = ({ title, sub, pct, lost, active, onClick, green }) => <button type="button" onClick={onClick} style={{ display: 'grid', gridTemplateColumns: '1fr 64px 16px', alignItems: 'center', gap: 10, width: '100%', padding: '7px 14px', background: active ? 'var(--ink-50)' : 'none', border: 0, borderTop: '1px solid var(--border-subtle)', cursor: 'pointer', textAlign: 'left', fontFamily: 'var(--font-sans)', color: 'var(--text-primary)' }}><span style={{ minWidth: 0 }}><span style={{ display: 'block', font: '600 13px/1.4 var(--font-sans)', whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>{title}</span><span style={{ display: 'block', font: '400 11px/1.4 var(--font-sans)', color: 'var(--text-secondary)', fontVariantNumeric: 'tabular-nums' }}>{sub}</span></span><span style={{ textAlign: 'right' }}><span style={{ display: 'block', font: '700 15px/1.2 var(--font-sans)', fontVariantNumeric: 'tabular-nums', color: green ? 'var(--green-800)' : 'inherit' }}>{pct}</span><span style={{ display: 'block', font: '500 10px/1.4 var(--font-sans)', color: lost > 0 ? 'var(--amber-800)' : 'var(--text-tertiary)', fontVariantNumeric: 'tabular-nums' }}>{lost > 0 ? '−' + lost + ' left' : ''}</span></span><span style={{ color: 'var(--text-tertiary)', fontSize: 18, lineHeight: 1 }}>›</span></button>;
          return <div style={{ display: 'grid', gap: 12 }}>
          <Card padding="wide" style={{ minWidth: 0, overflow: 'hidden' }}>
            <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', marginBottom: 6, gap: 12 }}>
              {zc ? <button type="button" onClick={() => { setZoom('all'); setSel(null); }} style={{ background: 'none', border: 0, padding: 0, cursor: 'pointer', font: '600 13px/1.4 var(--font-sans)', color: 'var(--ink-600)' }}>‹ All chapters</button> : <p style={{ margin: 0, font: '600 13px/1.4 var(--font-sans)' }}>{START} started · by chapter</p>}
              <p style={{ margin: 0, font: '700 13px/1.4 var(--font-sans)', color: 'var(--green-800)', fontVariantNumeric: 'tabular-nums', whiteSpace: 'nowrap' }}>{zc ? zc.label + ' · ' + Math.round(zc.exit / zc.enter * 100) + '% got through' : last.pct + '% paid'}</p>
            </div>
            <InkFunnelChart key={zoom} steps={zs} selected={sel} onSelect={id => zc ? setSel(id === sel ? null : id) : (setZoom(id), setSel(null))} showLabels height={zc ? 150 : 132} />
            <p style={{ margin: '4px 0 0', font: '400 11px/1.5 var(--font-sans)', color: 'var(--text-tertiary)' }}>{zc ? 'Each bar is one step · % of the ' + START + ' who started' : '% of the ' + START + ' still in at the end of each chapter · hatched = left'}</p>
          </Card>
          {!zc && <Card padding="none"><div style={{ marginTop: -1 }}>{chapters.map(c => <Row key={c.key} title={c.label} sub={c.enter + ' came in · ' + c.exit + ' finished · ' + c.steps.length + ' steps'} pct={Math.round(c.exit / c.enter * 100) + '%'} lost={c.enter - c.exit} green={c.key === 'paywall'} onClick={() => { setZoom(c.key); setSel(null); }} />)}</div></Card>}
          {zc && s && <div key={sel} style={{ animation: 'ink-fade var(--duration-base) var(--ease-out) both', display: 'grid', gap: 8 }}>
            <Card padding="wide">
              <p style={{ margin: 0, font: '500 11px/1.4 var(--font-sans)', color: 'var(--text-secondary)' }}>Step {f.indexOf(s) + 1} of {f.length} · <span style={{ fontFamily: 'var(--font-mono)' }}>{s.id}</span></p>
              <p style={{ margin: '2px 0 12px', font: '700 17px/1.3 var(--font-sans)', letterSpacing: '-0.01em' }}>{s.label}</p>
              <div style={{ display: 'grid', gridTemplateColumns: 'repeat(3,minmax(0,1fr))', gap: 8 }}>
                <Mini label="Got here" value={s.reached} sub={s.pct + '% of starters'} />
                <Mini label="Left here" value={s.dropped ? '−' + s.dropped : '0'} sub={s.dropPct + '% of arrivals'} />
                <Mini label="Went on" value={f[f.indexOf(s) + 1] ? Math.round(f[f.indexOf(s) + 1].reached / s.reached * 100) + '%' : '—'} sub="to next step" />
              </div>
            </Card>
            <div style={{ display: 'flex', gap: 8 }}>{f.indexOf(s) > 0 && <button type="button" onClick={() => { const p = f[f.indexOf(s) - 1]; setZoom(p.chapter); setSel(p.id); }} style={{ flex: 1, border: '1px solid var(--border-default)', background: 'var(--surface-card)', borderRadius: 12, padding: '10px 12px', cursor: 'pointer', font: '600 13px var(--font-sans)', color: 'var(--ink-700)', textAlign: 'left', minWidth: 0, whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>‹ {f[f.indexOf(s) - 1].label}</button>}{f[f.indexOf(s) + 1] && <button type="button" onClick={() => { const nx = f[f.indexOf(s) + 1]; setZoom(nx.chapter); setSel(nx.id); }} style={{ flex: 1, border: '1px solid var(--border-default)', background: 'var(--surface-card)', borderRadius: 12, padding: '10px 12px', cursor: 'pointer', font: '600 13px var(--font-sans)', color: 'var(--ink-700)', textAlign: 'right', minWidth: 0, whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>{f[f.indexOf(s) + 1].label} ›</button>}</div>
          </div>}
          {zc && !s && <Card padding="none"><p style={{ margin: 0, padding: '10px 14px 6px', font: '600 10px/1.4 var(--font-sans)', textTransform: 'uppercase', letterSpacing: '.04em', color: 'var(--text-tertiary)' }}>Where people leave this chapter</p>{top.map(st => <Row key={st.id} title={st.label} sub={st.reached + st.dropped + ' arrived · ' + st.reached + ' continued'} pct={st.dropPct + '%'} lost={st.dropped} onClick={() => setSel(st.id)} />)}</Card>}
        </div>; })()}
        {view === 'paywall' && (() => { const P = D.paywall, W = P.wheel; const pc = (a, b) => b ? Math.round(a / b * 100) : 0; const paidTotal = P.plans.reduce((s, p) => s + p.paid, 0), trialsTotal = P.plans.reduce((s, p) => s + p.trials, 0);
          const Step = ({ label, n, of, note, indent, green, warn, last }) => <div style={{ display: 'grid', gridTemplateColumns: '1fr 44px 44px', alignItems: 'center', gap: 8, padding: '7px 14px', paddingLeft: 14 + (indent || 0) * 14, borderTop: '1px solid var(--border-subtle)' }}><span style={{ minWidth: 0, display: 'flex', alignItems: 'center', gap: 8 }}>{indent ? <span style={{ width: 6, height: 6, borderRadius: 3, background: green ? 'var(--green-600)' : 'var(--ink-300)', flexShrink: 0 }} /> : null}<span style={{ minWidth: 0 }}><span style={{ display: 'block', font: `${indent ? 500 : 600} 13px/1.4 var(--font-sans)`, whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>{label}</span>{note && <span style={{ display: 'block', font: '400 10px/1.3 var(--font-sans)', color: 'var(--text-tertiary)' }}>{note}</span>}</span></span><span style={{ textAlign: 'right', font: '700 14px/1.2 var(--font-sans)', fontVariantNumeric: 'tabular-nums', color: green ? 'var(--green-800)' : 'inherit' }}>{n}</span><span style={{ textAlign: 'right', font: '500 11px/1.2 var(--font-sans)', fontVariantNumeric: 'tabular-nums', color: warn ? 'var(--amber-800)' : 'var(--text-tertiary)' }}>{of ? pc(n, of) + '%' : ''}</span></div>;
          return <div style={{ display: 'grid', gap: 12 }}>
          <Hero label="Saw the paywall → paid" value={pc(paidTotal, P.seen) + '%'} accent caption={paidTotal + ' of ' + P.seen + ' who hit the paywall · last 30 days'} right={<div style={{ textAlign: 'right', flexShrink: 0 }}><p style={{ margin: 0, font: '700 18px/1.2 var(--font-sans)', fontVariantNumeric: 'tabular-nums', color: 'var(--amber-800)' }}><CountUp value={P.stalled10m} /></p><p style={{ margin: 0, font: '600 10px/1.4 var(--font-sans)', textTransform: 'uppercase', letterSpacing: '.03em', color: 'var(--text-secondary)' }}>stalled 10 min+</p></div>} />
          <Card padding="none" style={{ overflow: 'hidden' }}>
            <div style={{ marginTop: -1 }}>
              <Step label="Saw the paywall" n={P.seen} note="% column = share of the row above it" />
              <Step label="Started a trial right away" n={P.trialDirect} of={P.seen} indent={1} green />
              <Step label="Closed the paywall" n={P.closed} of={P.seen} indent={1} warn />
              <Step label="Landed on the wheel" n={W.entered} of={P.closed} indent={2} />
              <Step label="Spun it" n={W.spun} of={W.entered} indent={2} />
              <Step label="Saw the 90% off screen" n={W.offer90} of={W.spun} indent={2} />
              <Step label="Started a trial from the offer" n={W.trial} of={W.offer90} indent={2} green />
              <Step label="Paid after the offer" n={W.paid} of={W.trial} indent={2} green />
            </div>
          </Card>
          <div>
            <SectionHeading>Trial → paid by plan</SectionHeading>
            <div style={{ display: 'grid', gridTemplateColumns: 'repeat(3,minmax(0,1fr))', gap: 8 }}>{P.plans.map(p => <Mini key={p.key} label={p.label} value={pc(p.paid, p.trials) + '%'} sub={p.paid + ' of ' + p.trials + ' trials'} accent={p.paid / p.trials === Math.max(...P.plans.map(x => x.paid / x.trials))} />)}</div>
          </div>
          <div style={{ display: 'grid', gridTemplateColumns: 'repeat(2,minmax(0,1fr))', gap: 8 }}>
            <Mini label="Wheel → paid" value={pc(W.paid, W.entered) + '%'} sub={W.paid + ' paid of ' + W.entered + ' who saw the wheel'} />
            <Mini label="Cancel save offer" value={pc(P.save.accepted, P.save.shown) + '%'} sub={P.save.accepted + ' of ' + P.save.shown + ' took the free month'} />
          </div>
        </div>; })()}
        {view === 'steps' && <div style={{ display: 'grid', gap: 12 }}>
          <div>
            <div style={{ display: 'flex', alignItems: 'center', justifyContent: 'space-between', marginBottom: 8 }}><SectionHeading style={{ marginBottom: 0 }}>Signups · {RATE_LABEL[range] === 'All' ? 'all time' : 'last ' + RATE_LABEL[range]}</SectionHeading>{rangeToggle}</div>
            <div style={{ display: 'grid', gridTemplateColumns: 'repeat(3,minmax(0,1fr))', gap: 8 }}>
              <Mini label="Started" value={started.length} sub={cohort.length - started.length + ' never started'} />
              <Mini label="Completed" value={completed.length} sub={(started.length ? Math.round(completed.length / started.length * 100) : 0) + '% of started'} />
              <Mini label="Stuck" value={inProg.length} sub="in progress" />
            </div>
          </div>
          <Card padding="none">
          {chapters.map((ch, ci) => { const open = !!openCh[ch.key]; const keep = ch.enter ? Math.round(ch.exit / ch.enter * 100) : 0; return (
            <div key={ch.key} style={{ borderTop: ci ? '1px solid var(--border-subtle)' : 0 }}>
              <button type="button" onClick={() => setOpenCh(o => ({ ...o, [ch.key]: !open }))} style={{ display: 'grid', gridTemplateColumns: '1fr auto 16px', alignItems: 'center', gap: 12, width: '100%', padding: '12px 16px', background: 'none', border: 0, cursor: 'pointer', textAlign: 'left', fontFamily: 'var(--font-sans)', color: 'var(--text-primary)' }}>
                <span><span style={{ display: 'block', font: '600 14px/1.4 var(--font-sans)' }}>{ch.label} <span style={{ font: '400 12px var(--font-sans)', color: 'var(--text-tertiary)' }}>· {ch.steps.length} steps</span></span><span style={{ display: 'block', marginTop: 6, height: 6, borderRadius: 3, background: 'var(--surface-track)', overflow: 'hidden' }}><span style={{ display: 'block', height: '100%', width: keep + '%', background: ch.key === 'paywall' ? 'var(--green-600)' : 'var(--ink-700)', borderRadius: 3, transformOrigin: '0 0', animation: 'ink-grow-x 600ms var(--ease-out) both' }} /></span></span>
                <span style={{ textAlign: 'right' }}><span style={{ display: 'block', font: '700 16px/1.2 var(--font-sans)', fontVariantNumeric: 'tabular-nums' }}>{keep}%</span><span style={{ font: '400 10px/1.4 var(--font-sans)', color: 'var(--text-tertiary)' }}>{ch.enter} → {ch.exit}</span></span>
                <span style={{ color: 'var(--text-tertiary)', transform: open ? 'rotate(90deg)' : 'none', transition: 'transform var(--duration-base) var(--ease-out)', fontSize: 18, lineHeight: 1 }}>›</span>
              </button>
              {open && <div style={{ animation: 'ink-fade var(--duration-base) var(--ease-out) both' }}>{ch.steps.map(st => <button key={st.id} type="button" onClick={() => goStep(st.id)} style={{ display: 'grid', gridTemplateColumns: '1fr 52px 52px', alignItems: 'center', gap: 10, width: '100%', padding: '8px 16px 8px 24px', background: 'transparent', border: 0, borderTop: '1px solid var(--border-subtle)', cursor: 'pointer', textAlign: 'left', fontFamily: 'var(--font-sans)', color: 'var(--text-primary)' }}><span style={{ minWidth: 0 }}><span style={{ display: 'block', font: '500 13px/1.4 var(--font-sans)', whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>{st.label}</span><span style={{ display: 'block', font: '400 10px/1.4 var(--font-mono)', color: 'var(--text-tertiary)' }}>{st.id}</span></span><span style={{ textAlign: 'right', font: '600 13px/1.4 var(--font-sans)', fontVariantNumeric: 'tabular-nums' }}>{st.pct}%</span><span style={{ textAlign: 'right', font: '500 12px/1.4 var(--font-sans)', fontVariantNumeric: 'tabular-nums', color: st.dropped > 10 ? 'var(--amber-800)' : 'var(--text-tertiary)' }}>{st.dropped ? '−' + st.dropped : ''}</span></button>)}</div>}
            </div>); })}
          </Card>
        </div>}
      </Fade>
    </div>
  );
}
Object.assign(window, { OnboardingTab });
